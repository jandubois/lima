func (*netConfig) restoreForwarding(context.Context) {}
func ipForwardEnabled() bool                         { return true }

var tables = map[string][][]string{} // "table chain" -> rules

func split(args []string) (key, op string, rule []string) {
	return args[2] + " " + args[4], args[3], args[5:]
}

func runOK(_ context.Context, _ string, args ...string) bool {
	key, _, rule := split(args)
	return slices.ContainsFunc(tables[key], func(r []string) bool { return slices.Equal(r, rule) })
}

func run(_ context.Context, name string, args ...string) error {
	if name != "iptables" {
		return nil
	}
	key, op, rule := split(args)
	switch op {
	case "-I":
		tables[key] = append([][]string{rule}, tables[key]...)
	case "-A":
		tables[key] = append(tables[key], rule)
	}
	return nil
}

func match(rule []string, in, out, src, dst string) (verdict string, ok bool) {
	neg := false
	for i := 0; i < len(rule); i++ {
		switch rule[i] {
		case "!":
			neg = true
			continue
		case "-j":
			return rule[i+1], true
		}
		var hit bool
		switch v := rule[i+1]; rule[i] {
		case "-i":
			hit = v == in
		case "-o":
			hit = v == out
		case "-s", "-d":
			_, n, _ := net.ParseCIDR(v)
			hit = n.Contains(net.ParseIP(map[string]string{"-s": src, "-d": dst}[rule[i]]))
		}
		if hit == neg {
			return "", false
		}
		neg = false
		i++
	}
	return "", false
}

func verdict(key, policy, in, out, src, dst string) string {
	for _, r := range tables[key] {
		if v, ok := match(r, in, out, src, dst); ok {
			return v
		}
	}
	return policy
}

Design Observations ¶

Concerns ¶

  • A starting instance's tap is not in use until QEMU opens it (future) Gemini 3.1 Pro Claude Opus 5.5 Codex GPT 6 Astra: ensureTaps creates the tap before the hostagent starts QEMU, and a new tap has no carrier. A concurrent last stop in another LIMA_HOME, or in this one before the starting instance counts as running, tears the network down and deletes the tap, and QEMU then fails to open it. The failure is loud, and it is round 1's unserialized-Reconcile concern; a host-wide lock in varRun around start and stop would close both. Counting every tap as in use would keep a network up forever, because taps outlive their instances.
  • stopNetwork still waits only 5 s (future) Claude Opus 5.5 Codex GPT 6 Astra: with the PID file removed last, a teardown whose firewall-cmd calls run past 5 s still overlaps a following start. CI exercises only the iptables branch, so nothing has measured it.
  • macOS keeps the cross-LIMA_HOME stop (future) Claude Opus 5.5: the guard covers lima-privileged-net only, so on macOS one LIMA_HOME stopping its last instance still stops socket_vmnet under another's.

Strengths ¶

  • Tap identity agrees across processes (in-scope) Claude Opus 5.5 Codex GPT 6 Astra: limactl and the hostagent both take the directory from store.Inspect, LimaDir resolves symlinks for both, and 100,000 generated IDs all pass identifiers.Validate.
  • The guard does not need sudo (in-scope) Claude Opus 5.5: it runs before validateConfig, so a LIMA_HOME that leaves a network alone never calls sudo.
  • The PID file goes last (in-scope) Codex GPT 6 Astra Claude Opus 5.5: stopNetwork now waits for the teardown, and an error from down() does not stop the removal.

Repo context updates ¶

  • [repo] On GitHub's hosted Ubuntu runners, Docker sets the iptables FORWARD policy to DROP, and the cloud network drops a reply addressed to an unmasqueraded private source. A negative connectivity test there ("guest X cannot reach Y") therefore passes whatever the firewall rules under test say. Flag such a test unless it also asserts that the rule exists, or has a positive control that fails when the rule is removed.
  • [repo] limactl stop -f and limactl delete -f reconcile networks right after StopForcibly sends SIGKILL, and nothing waits for the driver process to exit. A reconcile decision that reads live state the killed process holds (an open tap, a socket, a lock) can still see the dying instance. Flag such a check unless it excludes the reconciling LIMA_HOME's own instances.
