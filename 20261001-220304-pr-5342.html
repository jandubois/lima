pkg/sshutil/sshutil.go
// scp.exe and ssh-keygen.exe, which Lima needs for create and copy.
// MinGit ships ssh.exe without those, so a partial install is skipped for
// the next complete one on PATH, then the native install under
// %SystemRoot%\System32\OpenSSH. Returns "" when none qualifies, so the
// caller falls through to exec.LookPath instead of picking a partial one.
func pickCompleteSSHOnWindows() string {
	for _, dir := range filepath.SplitList(os.Getenv("PATH")) {
		if dir == "" {
			continue
		}
		sshPath := filepath.Join(dir, "ssh.exe")
		if _, err := os.Stat(sshPath); err != nil {
			continue
		}
		if missing := missingSiblings(dir, "scp.exe", "ssh-keygen.exe"); len(missing) > 0 {
			logrus.Debugf("skipping ssh at %#q: missing %v in same directory (likely MinGit or another partial install)", sshPath, missing)
			continue
		}
		return sshPath
	}
	nativeDir := filepath.Join(systemRoot(), "System32", "OpenSSH")
	nativeSSH := filepath.Join(nativeDir, "ssh.exe")
	if _, err := os.Stat(nativeSSH); err == nil {
		if missing := missingSiblings(nativeDir, "scp.exe", "ssh-keygen.exe"); len(missing) > 0 {
			logrus.Debugf("skipping native ssh at %#q: missing %v in same directory", nativeSSH, missing)
			return ""
		}
		return nativeSSH
	}

	return ""
}

Review Process Notes ¶

Skill improvements ¶

  • Fetch the head commit's CI results before launching agents, not only before consolidating, and put each failing job's error excerpt, with the same job's outcome at the merge-base, into the prompt. Agents cannot reach CI themselves. With the failure in hand they spend the round on its cause and its fix, and a failure first seen at consolidation gets the orchestrator's analysis alone.
  • When a change alters what runs before a test step (installs a tool, moves setup ahead of the test command, changes PATH), list the tests that skip when that tool is absent and check them against the new environment. A skip guard keyed on a tool's presence turns an environment change into newly running code the diff does not show.

Repo context updates ¶

  • [repo] On Windows, limactl appends %PROGRAMFILES%\QEMU and C:\msys64\usr\bin to its own PATH whenever those directories exist (envutil.PlatformCommonToolDirs, applied in cmd/limactl/main.go). Removing a toolchain from PATH therefore does not hide it from limactl; only deleting the directory does.
